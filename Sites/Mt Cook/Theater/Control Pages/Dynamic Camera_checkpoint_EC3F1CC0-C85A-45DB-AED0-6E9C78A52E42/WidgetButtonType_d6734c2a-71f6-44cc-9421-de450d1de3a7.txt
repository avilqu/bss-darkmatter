	DigitalSky.LogMessage("Checkpoint 1");
	DigitalSky.Wait(1);

	camTar.SetProperty("Parent", mainCamParent, 0);

	DigitalSky.LogMessage("mainCamParent Value is: " + mainCamParent + ". ");
